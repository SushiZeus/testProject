TAX PAID FIX (DeclarationPage.tsx):
────────────────────────────────────
const handleTaxPaid = () => {
  if (!selectedFile || !user) return;
  
  // NEW: Determine new status
  const newStatus = selectedFile.transportMode === 'SEA'
    ? 'WAITING_FOR_WHARFAGE_PAYMENT'  // ✅ SEA: Next is wharfage
    : 'TAXES_PAID';  // ✅ AIR/ROAD: Tax only, mark as paid
  
  updateFileStatus(
    selectedFile.id,
    newStatus,  // ✅ CHANGE STATUS
    user.id,
    {
      taxPaymentConfirmed: true,
      taxPaymentConfirmedAt: new Date(),
    }
  );
  
  toast.success('Tax payment confirmed ✓');
  setSelectedFile(null);
};
